## 1. 概述与需求分析

### 1.1 核心挑战

| 挑战 | 说明 |
|------|------|
| **浏览器内存限制** | 一次性读取 >100MB 文件会导致标签页崩溃 |
| **网络中断** | 上传大文件过程中网络断开需从头重传 |
| **服务器限制** | Nginx/Node 默认请求体大小限制（通常 1MB~50MB） |
| **上传超时** | 慢速网络下大文件上传可能超时断开 |
| **并发控制** | 过多分片同时上传会耗尽浏览器连接数（HTTP/1.1 仅 6 个） |
| **用户体验** | 缺乏进度反馈、无法暂停/恢复 |

### 1.2 方案选型矩阵

```
                    是否需要断点续传？
                   /                \
                 是                  否
                /                      \
        文件是否超 500MB？         是否需要流式处理？
         /          \              /              \
       是            否           是               否
       |             |            |                |
   方案二          方案一       方案三           方案一
 (断点续传)    (分片上传+hash)  (流式上传)    (基础分片上传)
```

### 1.3 核心技术对比预览

| 维度 | 分片上传 | 断点续传 | 流式上传 |
|------|----------|----------|----------|
| 核心API | `Blob.slice()` | `Blob.slice()` + SparkMD5 | `File.stream()` / `ReadableStream` |
| 记忆体占用 | 低（分片级） | 低（分片级） | 极低（流式） |
| 断网恢复 | 需手动实现 | 自动恢复 | 需手动实现 |
| 实现复杂度 | 中等 | 较高 | 中等 |
| 浏览器兼容 | IE10+ | IE10+ | Chrome 78+ / Edge 79+ |

## 3. 方案一：分片上传

### 3.1 原理

将大文件通过 `Blob.slice()` 切割成固定大小的分片（如 5MB），逐片上传到服务端，全部上传完成后服务端将分片合并为完整文件。

```
┌──────┬──────┬──────┬──────┬──────┬──────┐
│Chunk0│Chunk1│Chunk2│Chunk3│Chunk4│Chunk5│
│ 5MB  │ 5MB  │ 5MB  │ 5MB  │ 5MB  │ 2MB  │
└──┬───┴──┬───┴──┬───┴──┬───┴──┬───┴──┬───┘
   │      │      │      │      │      │
   ▼      ▼      ▼      ▼      ▼      ▼
┌─────────────────────────────────────────┐
│          服务端：分片存储 → 合并          │
│  /tmp/chunks/fileId/                     │
│    ├── chunk_0                           │
│    ├── chunk_1                           │
│    ├── ...                               │
│    └── chunk_5                           │
│                                          │
│  合并 → /uploads/fileId/originalName     │
└─────────────────────────────────────────┘
```

### 3.2 前端实现

```javascript
/**
 * 方案一：分片上传器
 *
 * 特性：
 *  - 文件分片（Blob.slice）
 *  - 并发控制（同时最多 N 个分片）
 *  - 失败自动重试（每个分片最多重试 3 次）
 *  - 实时进度反馈
 *  - 支持暂停/恢复
 *  - 上传完成后触发合并
 */
class ChunkedUploader {
    /**
     * @param {Object} options
     * @param {File}   options.file          - 要上传的文件
     * @param {number} [options.chunkSize=5*1024*1024]  - 分片大小（默认 5MB）
     * @param {number} [options.concurrency=3] - 并发上传数
     * @param {number} [options.maxRetries=3]  - 每个分片最大重试次数
     * @param {string} options.uploadUrl       - 分片上传接口
     * @param {string} options.mergeUrl        - 合并接口
     * @param {Function} [options.onProgress]  - 进度回调 (percent, uploaded, total)
     * @param {Function} [options.onSuccess]   - 成功回调 (result)
     * @param {Function} [options.onError]     - 错误回调 (error)
     * @param {Function} [options.onChunkComplete] - 单个分片完成回调 (index, total)
     */
    constructor(options) {
        this.file = options.file;
        this.chunkSize = options.chunkSize || 5 * 1024 * 1024; // 5MB
        this.concurrency = options.concurrency || 3;
        this.maxRetries = options.maxRetries || 3;
        this.uploadUrl = options.uploadUrl;
        this.mergeUrl = options.mergeUrl;

        // 回调
        this.onProgress = options.onProgress || (() => {});
        this.onSuccess = options.onSuccess || (() => {});
        this.onError = options.onError || (() => {});
        this.onChunkComplete = options.onChunkComplete || (() => {});

        // 状态
        this.status = UploadStatus.PENDING;
        this.abortController = new AbortController();

        // 分片信息
        this.totalChunks = Math.ceil(this.file.size / this.chunkSize);
        this.chunks = this.createChunks();

        // 进度追踪
        this.completedChunks = new Set();
        this.failedChunks = new Map(); // index → retryCount
        this.uploadedBytes = 0;

        // 暂停队列
        this.pausedQueue = [];
    }

    /**
     * 创建分片元数据数组
     */
    createChunks() {
        const chunks = [];
        for (let i = 0; i < this.totalChunks; i++) {
            const start = i * this.chunkSize;
            const end = Math.min(start + this.chunkSize, this.file.size);
            chunks.push({
                index: i,
                start,
                end,
                size: end - start,
                blob: this.file.slice(start, end)
            });
        }
        return chunks;
    }

    /**
     * 开始上传
     */
    async start() {
        if (this.status === UploadStatus.UPLOADING) return;

        this.status = UploadStatus.UPLOADING;
        this.abortController = new AbortController();

        try {
            // 创建并发控制池
            const pool = new Array(this.concurrency).fill(null);
            let chunkIndex = 0;

            const uploadNext = async () => {
                while (chunkIndex < this.totalChunks && this.status === UploadStatus.UPLOADING) {
                    const idx = chunkIndex++;
                    if (this.completedChunks.has(idx)) continue;

                    await this.uploadChunk(idx);
                }
            };

            // 启动并发上传
            await Promise.all(pool.map(() => uploadNext()));

            // 所有分片完成后触发合并
            if (this.status === UploadStatus.UPLOADING) {
                await this.merge();
            }
        } catch (error) {
            if (error.name !== 'AbortError') {
                this.status = UploadStatus.FAILED;
                this.onError(error);
            }
        }
    }

    /**
     * 上传单个分片
     */
    async uploadChunk(index) {
        const chunk = this.chunks[index];
        const formData = new FormData();
        formData.append('file', chunk.blob, `chunk_${index}`);
        formData.append('index', String(index));
        formData.append('total', String(this.totalChunks));
        formData.append('fileName', this.file.name);
        formData.append('fileSize', String(this.file.size));
        formData.append('fileId', this.fileId); // 由服务端首次返回

        let retries = this.failedChunks.get(index) || 0;

        while (retries <= this.maxRetries) {
            try {
                const response = await fetch(this.uploadUrl, {
                    method: 'POST',
                    body: formData,
                    signal: this.abortController.signal
                });

                if (!response.ok) {
                    throw new Error(`分片 ${index} 上传失败: ${response.status}`);
                }

                const result = await response.json();

                // 保存服务端返回的 fileId
                if (!this.fileId && result.data.fileId) {
                    this.fileId = result.data.fileId;
                }

                // 标记完成
                this.completedChunks.add(index);
                this.uploadedBytes += chunk.size;
                this.failedChunks.delete(index);

                // 进度回调
                const progress = Math.round((this.completedChunks.size / this.totalChunks) * 100);
                this.onProgress(progress, this.uploadedBytes, this.file.size);
                this.onChunkComplete(index, this.totalChunks);

                return;
            } catch (error) {
                if (error.name === 'AbortError') throw error;

                retries++;
                this.failedChunks.set(index, retries);

                if (retries > this.maxRetries) {
                    throw new Error(`分片 ${index} 重试 ${this.maxRetries} 次后仍失败`);
                }

                // 指数退避
                await this.delay(Math.pow(2, retries) * 1000);
            }
        }
    }

    /**
     * 通知服务端合并分片
     */
    async merge() {
        this.status = UploadStatus.MERGING;

        try {
            const response = await fetch(this.mergeUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    fileId: this.fileId,
                    fileName: this.file.name,
                    fileSize: this.file.size,
                    totalChunks: this.totalChunks
                }),
                signal: this.abortController.signal
            });

            if (!response.ok) {
                throw new Error(`合并失败: ${response.status}`);
            }

            const result = await response.json();
            this.status = UploadStatus.COMPLETED;
            this.onProgress(100, this.file.size, this.file.size);
            this.onSuccess(result);
        } catch (error) {
            if (error.name !== 'AbortError') {
                this.status = UploadStatus.FAILED;
                this.onError(error);
            }
        }
    }

    /**
     * 暂停上传
     */
    pause() {
        this.status = UploadStatus.PAUSED;
        this.abortController.abort();
    }

    /**
     * 恢复上传（从暂停位置继续）
     */
    async resume() {
        this.status = UploadStatus.UPLOADING;
        this.abortController = new AbortController();
        await this.start();
    }

    /**
     * 取消上传
     */
    cancel() {
        this.status = UploadStatus.FAILED;
        this.abortController.abort();
        this.completedChunks.clear();
        this.uploadedBytes = 0;
    }

    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}

// ==================== 使用示例 ====================
function createChunkedUploadDemo() {
    const fileInput = document.getElementById('fileInput');
    const progressBar = document.getElementById('progressBar');
    const progressText = document.getElementById('progressText');
    const pauseBtn = document.getElementById('pauseBtn');
    const resumeBtn = document.getElementById('resumeBtn');
    const cancelBtn = document.getElementById('cancelBtn');

    let uploader = null;

    fileInput.addEventListener('change', (e) => {
        const file = e.target.files[0];
        if (!file) return;

        uploader = new ChunkedUploader({
            file,
            chunkSize: 5 * 1024 * 1024,   // 5MB
            concurrency: 3,                 // 同时 3 个分片
            maxRetries: 3,
            uploadUrl: '/api/upload/chunk',
            mergeUrl: '/api/upload/merge',
            onProgress: (percent, uploaded, total) => {
                progressBar.style.width = percent + '%';
                progressText.textContent =
                    `${percent}% (${formatSize(uploaded)} / ${formatSize(total)})`;
            },
            onSuccess: (result) => {
                console.log('上传成功:', result);
                progressText.textContent = '上传完成!';
            },
            onError: (error) => {
                console.error('上传失败:', error);
                progressText.textContent = '上传失败: ' + error.message;
            },
            onChunkComplete: (index, total) => {
                console.log(`分片 ${index + 1}/${total} 完成`);
            }
        });

        uploader.start();
    });

    pauseBtn.addEventListener('click', () => uploader?.pause());
    resumeBtn.addEventListener('click', () => uploader?.resume());
    cancelBtn.addEventListener('click', () => uploader?.cancel());
}

function formatSize(bytes) {
    if (bytes < 1024) return bytes + ' B';
    if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' KB';
    if (bytes < 1073741824) return (bytes / 1048576).toFixed(1) + ' MB';
    return (bytes / 1073741824).toFixed(2) + ' GB';
}
```

### 3.3 服务端实现（Node.js）

```javascript
const express = require('express');
const multer = require('multer');
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

// ==================== 配置 ====================
const CHUNK_DIR = path.join(__dirname, 'chunks');
const UPLOAD_DIR = path.join(__dirname, 'uploads');

// 确保目录存在
[CHUNK_DIR, UPLOAD_DIR].forEach(dir => {
    if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
});

// 配置文件上传
const upload = multer({
    storage: multer.diskStorage({
        destination: (req, file, cb) => {
            // 分片存放到临时目录，按 fileId 分组
            const fileId = req.body.fileId || crypto.randomUUID();
            const chunkDir = path.join(CHUNK_DIR, fileId);
            if (!fs.existsSync(chunkDir)) fs.mkdirSync(chunkDir, { recursive: true });
            cb(null, chunkDir);
        },
        filename: (req, file, cb) => {
            // 分片命名为 chunk_序号
            cb(null, `chunk_${req.body.index}`);
        }
    }),
    limits: {
        fileSize: 10 * 1024 * 1024 // 单个分片最大 10MB
    }
});

const app = express();
app.use(express.json());

// ==================== 接口 ====================

/**
 * POST /api/upload/chunk
 * 接收单个分片
 *
 * 请求体（multipart/form-data）：
 *   file: Blob           - 分片文件数据
 *   index: number        - 分片索引
 *   total: number        - 总分片数
 *   fileName: string     - 原始文件名
 *   fileSize: number     - 文件总大小
 *   fileId: string?      - 文件唯一标识（首次为空，后续由服务端返回）
 *
 * 响应：
 *   { code: 0, data: { fileId: string, index: number } }
 */
app.post('/api/upload/chunk', upload.single('file'), (req, res) => {
    const { index, total, fileName, fileSize } = req.body;
    let fileId = req.body.fileId;

    // 首次上传，生成 fileId
    if (!fileId) {
        fileId = crypto.createHash('md5')
            .update(`${fileName}_${fileSize}_${Date.now()}`)
            .digest('hex');
    }

    console.log(`[分片上传] fileId=${fileId}, chunk=${index}/${total}`);

    res.json({
        code: 0,
        message: '分片上传成功',
        data: { fileId, index: Number(index) }
    });
});

/**
 * POST /api/upload/merge
 * 合并所有分片为完整文件
 *
 * 请求体（JSON）：
 *   { fileId: string, fileName: string, fileSize: number, totalChunks: number }
 *
 * 响应：
 *   { code: 0, data: { fileId, fileName, url } }
 */
app.post('/api/upload/merge', async (req, res) => {
    const { fileId, fileName, fileSize, totalChunks } = req.body;

    const chunkDir = path.join(CHUNK_DIR, fileId);
    const targetPath = path.join(UPLOAD_DIR, `${fileId}_${fileName}`);

    try {
        // 校验分片完整性
        for (let i = 0; i < totalChunks; i++) {
            const chunkPath = path.join(chunkDir, `chunk_${i}`);
            if (!fs.existsSync(chunkPath)) {
                return res.status(400).json({
                    code: 1,
                    message: `分片 ${i} 缺失，请重新上传`
                });
            }
        }

        // 创建写入流
        const writeStream = fs.createWriteStream(targetPath);

        // 按顺序合并分片
        for (let i = 0; i < totalChunks; i++) {
            const chunkPath = path.join(chunkDir, `chunk_${i}`);
            const chunkData = fs.readFileSync(chunkPath);
            writeStream.write(chunkData);
        }

        writeStream.end();

        // 等待写入完成
        await new Promise((resolve, reject) => {
            writeStream.on('finish', resolve);
            writeStream.on('error', reject);
        });

        // 校验文件大小
        const stat = fs.statSync(targetPath);
        if (stat.size !== Number(fileSize)) {
            throw new Error(`文件大小校验失败: 期望 ${fileSize}, 实际 ${stat.size}`);
        }

        // 清理分片目录
        fs.rmSync(chunkDir, { recursive: true, force: true });

        console.log(`[合并完成] fileId=${fileId}, ${fileName}`);

        res.json({
            code: 0,
            message: '合并成功',
            data: {
                fileId,
                fileName,
                url: `/uploads/${fileId}_${fileName}`,
                size: stat.size
            }
        });
    } catch (error) {
        console.error('[合并失败]', error);
        res.status(500).json({
            code: 1,
            message: error.message
        });
    }
});

// 静态文件服务
app.use('/uploads', express.static(UPLOAD_DIR));

app.listen(3000, () => {
    console.log('分片上传服务启动: http://localhost:3000');
});
```

## 4. 方案二：断点续传

### 4.1 原理

在分片上传基础上，增加**文件 hash 计算**和**已上传分片查询**机制。上传前先计算文件 hash 并查询服务端已有哪些分片，只上传缺失的分片，实现真正的断点续传。

```
协议流程：
┌──────────┐                              ┌──────────┐
│  客户端   │                              │  服务端   │
└────┬─────┘                              └────┬─────┘
     │                                         │
     │  POST /api/upload/init                  │
     │  { fileName, fileSize, fileHash }       │
     │ ──────────────────────────────────────► │
     │                                         │ 计算文件 hash
     │  { fileId, uploadedChunks: [0,1,3] }    │ 检查已传分片
     │ ◄────────────────────────────────────── │
     │                                         │
     │  POST /api/upload/chunk/{fileId}        │
     │  (只上传缺失的分片: 2, 4, 5...)          │ → 存储分片
     │ ──────────────────────────────────────► │
     │                                         │
     │  POST /api/upload/merge/{fileId}        │
     │ ──────────────────────────────────────► │ → 合并文件
     │                                         │ → 校验 hash
     │  { url, size }                          │
     │ ◄────────────────────────────────────── │
     │
     │   如果中途断网，重新连接后：
     │   重新 Init → 服务端返回已上传分片 → 跳过已完成的分片，继续上传
```

### 4.2 前端实现

```javascript
/**
 * 方案二：断点续传上传器
 *
 * 核心增强：
 *  - 使用 SparkMD5 计算文件 hash（增量计算，避免一次性读取整个文件）
 *  - 上传前查询服务端已上传的分片
 *  - 只上传缺失的分片
 *  - 支持页面刷新后恢复上传（通过 localStorage 持久化状态）
 *  - 分片级 hash 校验确保数据完整性
 */
class ResumableUploader {
    /**
     * @param {Object} options
     * @param {File}   options.file
     * @param {string} options.baseUrl         - API 基础路径
     * @param {number} [options.chunkSize=5*1024*1024]
     * @param {number} [options.concurrency=3]
     * @param {Function} [options.onProgress]
     * @param {Function} [options.onSuccess]
     * @param {Function} [options.onError]
     * @param {Function} [options.onHashProgress] - hash 计算进度回调
     */
    constructor(options) {
        this.file = options.file;
        this.baseUrl = options.baseUrl;
        this.chunkSize = options.chunkSize || 5 * 1024 * 1024;
        this.concurrency = options.concurrency || 3;

        this.onProgress = options.onProgress || (() => {});
        this.onSuccess = options.onSuccess || (() => {});
        this.onError = options.onError || (() => {});
        this.onHashProgress = options.onHashProgress || (() => {});

        // 状态
        this.status = UploadStatus.PENDING;
        this.fileId = null;
        this.fileHash = null;
        this.totalChunks = Math.ceil(this.file.size / this.chunkSize);
        this.uploadedChunks = new Set();
        this.abortController = null;

        // 持久化 key
        this.storageKey = `upload_${this.file.name}_${this.file.size}`;
    }

    /**
     * 计算文件 MD5 hash（增量计算，避免内存溢出）
     */
    async computeHash() {
        return new Promise((resolve, reject) => {
            const spark = new SparkMD5.ArrayBuffer();
            const reader = new FileReader();
            let offset = 0;

            reader.onload = (e) => {
                spark.append(e.target.result);
                offset += this.chunkSize;

                const progress = Math.min(100, Math.round((offset / this.file.size) * 100));
                this.onHashProgress(progress);

                if (offset < this.file.size) {
                    readNext();
                } else {
                    const hash = spark.end();
                    resolve(hash);
                }
            };

            reader.onerror = () => reject(new Error('文件读取失败'));

            function readNext() {
                const blob = this.file.slice(offset, offset + this.chunkSize);
                reader.readAsArrayBuffer(blob);
            }

            readNext = readNext.bind(this);
            readNext();
        });
    }

    /**
     * 初始化上传（查询已上传分片）
     */
    async init() {
        this.status = UploadStatus.UPLOADING;

        // 1. 计算文件 hash
        console.log('正在计算文件 hash...');
        this.fileHash = await this.computeHash();
        console.log('文件 hash:', this.fileHash);

        // 2. 查询服务端已上传的分片
        const response = await fetch(`${this.baseUrl}/api/upload/init`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                fileName: this.file.name,
                fileSize: this.file.size,
                fileHash: this.fileHash,
                totalChunks: this.totalChunks
            })
        });

        const result = await response.json();
        this.fileId = result.data.fileId;

        // 服务端返回已上传的分片列表
        if (result.data.uploadedChunks) {
            result.data.uploadedChunks.forEach(idx => this.uploadedChunks.add(idx));
        }

        // 如果所有分片已上传，直接合并
        if (this.uploadedChunks.size >= this.totalChunks) {
            await this.merge();
            return;
        }

        console.log(`已上传 ${this.uploadedChunks.size}/${this.totalChunks} 分片，继续上传...`);

        // 持久化状态
        this.saveState();

        // 3. 开始上传缺失的分片
        await this.uploadMissingChunks();
    }

    /**
     * 上传缺失的分片
     */
    async uploadMissingChunks() {
        this.abortController = new AbortController();

        // 获取需要上传的分片索引
        const pendingChunks = [];
        for (let i = 0; i < this.totalChunks; i++) {
            if (!this.uploadedChunks.has(i)) {
                pendingChunks.push(i);
            }
        }

        console.log(`待上传分片: ${pendingChunks.length}`);

        // 并发控制
        let activeCount = 0;
        let currentIndex = 0;

        const uploadNext = async () => {
            while (currentIndex < pendingChunks.length && this.status === UploadStatus.UPLOADING) {
                const chunkIndex = pendingChunks[currentIndex++];
                activeCount++;

                try {
                    await this.uploadChunk(chunkIndex);
                    this.uploadedChunks.add(chunkIndex);
                    this.saveState();

                    const progress = Math.round(
                        (this.uploadedChunks.size / this.totalChunks) * 100
                    );
                    this.onProgress(progress, this.uploadedChunks.size, this.totalChunks);
                } catch (err) {
                    if (err.name === 'AbortError') return;
                    console.error(`分片 ${chunkIndex} 上传失败:`, err);
                    // 失败的分片重新加入队列
                    pendingChunks.push(chunkIndex);
                } finally {
                    activeCount--;
                }
            }
        };

        // 启动并发上传
        const workers = [];
        for (let i = 0; i < this.concurrency; i++) {
            workers.push(uploadNext());
        }

        await Promise.all(workers);

        // 检查是否全部完成
        if (this.uploadedChunks.size >= this.totalChunks) {
            this.clearState();
            await this.merge();
        }
    }

    /**
     * 上传单个分片
     */
    async uploadChunk(index) {
        const start = index * this.chunkSize;
        const end = Math.min(start + this.chunkSize, this.file.size);
        const blob = this.file.slice(start, end);

        const formData = new FormData();
        formData.append('chunk', blob, `chunk_${index}`);
        formData.append('index', String(index));
        formData.append('total', String(this.totalChunks));
        formData.append('fileHash', this.fileHash);

        const response = await fetch(
            `${this.baseUrl}/api/upload/chunk/${this.fileId}`,
            {
                method: 'POST',
                body: formData,
                signal: this.abortController.signal
            }
        );

        if (!response.ok) {
            throw new Error(`分片 ${index} 上传失败: ${response.status}`);
        }

        return response.json();
    }

    /**
     * 合并分片
     */
    async merge() {
        this.status = UploadStatus.MERGING;

        const response = await fetch(`${this.baseUrl}/api/upload/merge/${this.fileId}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                fileHash: this.fileHash,
                totalChunks: this.totalChunks
            })
        });

        if (!response.ok) {
            throw new Error(`合并失败: ${response.status}`);
        }

        const result = await response.json();
        this.status = UploadStatus.COMPLETED;
        this.clearState();
        this.onProgress(100, this.totalChunks, this.totalChunks);
        this.onSuccess(result.data);
    }

    // ==================== 状态持久化 ====================

    saveState() {
        try {
            localStorage.setItem(this.storageKey, JSON.stringify({
                fileId: this.fileId,
                fileHash: this.fileHash,
                uploadedChunks: Array.from(this.uploadedChunks),
                totalChunks: this.totalChunks,
                timestamp: Date.now()
            }));
        } catch (e) {
            // localStorage 可能已满，忽略
        }
    }

    clearState() {
        localStorage.removeItem(this.storageKey);
    }

    restoreState() {
        try {
            const data = localStorage.getItem(this.storageKey);
            if (!data) return false;

            const state = JSON.parse(data);
            // 过期检测（超过 24 小时）
            if (Date.now() - state.timestamp > 24 * 60 * 60 * 1000) {
                this.clearState();
                return false;
            }

            this.fileId = state.fileId;
            this.fileHash = state.fileHash;
            this.uploadedChunks = new Set(state.uploadedChunks);
            this.totalChunks = state.totalChunks;
            return true;
        } catch (e) {
            return false;
        }
    }

    // ==================== 控制方法 ====================

    async start() {
        // 尝试恢复之前的状态
        const restored = this.restoreState();

        if (restored) {
            console.log('恢复上传状态，已上传:', this.uploadedChunks.size);
            this.status = UploadStatus.UPLOADING;
            await this.uploadMissingChunks();
        } else {
            await this.init();
        }
    }

    pause() {
        this.status = UploadStatus.PAUSED;
        if (this.abortController) {
            this.abortController.abort();
        }
    }

    async resume() {
        this.status = UploadStatus.UPLOADING;
        await this.uploadMissingChunks();
    }

    cancel() {
        this.status = UploadStatus.FAILED;
        if (this.abortController) {
            this.abortController.abort();
        }
        this.clearState();
    }
}
```

```javascript
const express = require('express');
const multer = require('multer');
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

const CHUNK_DIR = path.join(__dirname, 'chunks');
const UPLOAD_DIR = path.join(__dirname, 'uploads');
const META_DIR = path.join(__dirname, 'meta'); // 上传元数据存储

[CHUNK_DIR, UPLOAD_DIR, META_DIR].forEach(dir => {
    if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
});

// 文件上传元数据管理
class UploadMetaManager {
    getMetaPath(fileId) {
        return path.join(META_DIR, `${fileId}.json`);
    }

    save(fileId, meta) {
        fs.writeFileSync(this.getMetaPath(fileId), JSON.stringify(meta));
    }

    load(fileId) {
        const metaPath = this.getMetaPath(fileId);
        return fs.existsSync(metaPath)
            ? JSON.parse(fs.readFileSync(metaPath, 'utf-8'))
            : null;
    }

    delete(fileId) {
        const metaPath = this.getMetaPath(fileId);
        if (fs.existsSync(metaPath)) fs.unlinkSync(metaPath);
    }

    /**
     * 根据 fileHash 查找已有的 fileId（去重）
     */
    findByHash(fileHash) {
        if (!fs.existsSync(META_DIR)) return null;
        const files = fs.readdirSync(META_DIR);
        for (const file of files) {
            if (!file.endsWith('.json')) continue;
            const meta = JSON.parse(fs.readFileSync(path.join(META_DIR, file), 'utf-8'));
            if (meta.fileHash === fileHash) {
                return meta;
            }
        }
        return null;
    }

    /**
     * 获取已上传的分片索引列表
     */
    getUploadedChunks(fileId) {
        const chunkDir = path.join(CHUNK_DIR, fileId);
        if (!fs.existsSync(chunkDir)) return [];

        return fs.readdirSync(chunkDir)
            .filter(f => f.startsWith('chunk_'))
            .map(f => parseInt(f.replace('chunk_', '')))
            .sort((a, b) => a - b);
    }
}

const metaManager = new UploadMetaManager();

// 配置文件上传
const chunkUpload = multer({
    storage: multer.diskStorage({
        destination: (req, file, cb) => {
            const fileId = req.params.fileId;
            const chunkDir = path.join(CHUNK_DIR, fileId);
            if (!fs.existsSync(chunkDir)) fs.mkdirSync(chunkDir, { recursive: true });
            cb(null, chunkDir);
        },
        filename: (req, file, cb) => {
            cb(null, `chunk_${req.body.index}`);
        }
    }),
    limits: { fileSize: 10 * 1024 * 1024 }
});

const app = express();
app.use(express.json());

// ==================== 接口 ====================

/**
 * POST /api/upload/init
 * 初始化上传，返回已上传的分片列表
 */
app.post('/api/upload/init', (req, res) => {
    const { fileName, fileSize, fileHash, totalChunks } = req.body;

    // 检查是否有相同 hash 的上传记录（去重）
    const existing = metaManager.findByHash(fileHash);
    let fileId;

    if (existing) {
        // 复用已有的 fileId
        fileId = existing.fileId;
        console.log(`[Init] 复用已有上传记录: ${fileId}`);
    } else {
        // 生成新的 fileId
        fileId = crypto.createHash('md5')
            .update(`${fileHash}_${Date.now()}`)
            .digest('hex');

        metaManager.save(fileId, {
            fileId,
            fileName,
            fileSize,
            fileHash,
            totalChunks,
            createdAt: Date.now()
        });
    }

    // 获取已上传的分片
    const uploadedChunks = metaManager.getUploadedChunks(fileId);

    res.json({
        code: 0,
        data: {
            fileId,
            uploadedChunks,
            totalChunks
        }
    });
});

/**
 * POST /api/upload/chunk/:fileId
 * 上传单个分片
 */
app.post('/api/upload/chunk/:fileId', chunkUpload.single('chunk'), (req, res) => {
    const { fileId } = req.params;
    const { index, fileHash } = req.body;

    // 更新元数据中的 hash（确保一致）
    const meta = metaManager.load(fileId);
    if (meta && !meta.fileHash) {
        meta.fileHash = fileHash;
        metaManager.save(fileId, meta);
    }

    // 校验分片 hash
    const chunkPath = path.join(CHUNK_DIR, fileId, `chunk_${index}`);
    const chunkHash = crypto.createHash('md5')
        .update(fs.readFileSync(chunkPath))
        .digest('hex');

    console.log(`[Chunk] fileId=${fileId}, index=${index}, hash=${chunkHash}`);

    res.json({
        code: 0,
        data: { index: Number(index) }
    });
});

/**
 * POST /api/upload/merge/:fileId
 * 合并分片
 */
app.post('/api/upload/merge/:fileId', async (req, res) => {
    const { fileId } = req.params;
    const { fileHash, totalChunks } = req.body;

    const meta = metaManager.load(fileId);
    if (!meta) {
        return res.status(404).json({ code: 1, message: '上传记录不存在' });
    }

    const chunkDir = path.join(CHUNK_DIR, fileId);
    const targetPath = path.join(UPLOAD_DIR, `${fileId}_${meta.fileName}`);

    try {
        // 校验分片完整性
        const uploadedChunks = metaManager.getUploadedChunks(fileId);
        if (uploadedChunks.length !== totalChunks) {
            return res.status(400).json({
                code: 1,
                message: `分片缺失: 已有 ${uploadedChunks.length}/${totalChunks}`
            });
        }

        // 合并分片
        const writeStream = fs.createWriteStream(targetPath);
        for (let i = 0; i < totalChunks; i++) {
            const chunkPath = path.join(chunkDir, `chunk_${i}`);
            writeStream.write(fs.readFileSync(chunkPath));
        }
        writeStream.end();

        await new Promise((resolve, reject) => {
            writeStream.on('finish', resolve);
            writeStream.on('error', reject);
        });

        // 校验完整文件 hash
        const hash = crypto.createHash('md5');
        const fileStream = fs.createReadStream(targetPath);
        fileStream.on('data', chunk => hash.update(chunk));
        await new Promise((resolve) => fileStream.on('end', resolve));

        const finalHash = hash.digest('hex');
        console.log(`[Merge] fileId=${fileId}, finalHash=${finalHash}`);

        // 清理分片和元数据
        fs.rmSync(chunkDir, { recursive: true, force: true });
        metaManager.delete(fileId);

        res.json({
            code: 0,
            data: {
                fileId,
                fileHash: finalHash,
                url: `/uploads/${fileId}_${meta.fileName}`,
                size: fs.statSync(targetPath).size
            }
        });
    } catch (error) {
        console.error('[Merge Error]', error);
        res.status(500).json({ code: 1, message: error.message });
    }
});

/**
 * GET /api/upload/status/:fileId
 * 查询上传进度
 */
app.get('/api/upload/status/:fileId', (req, res) => {
    const { fileId } = req.params;
    const meta = metaManager.load(fileId);

    if (!meta) {
        return res.status(404).json({ code: 1, message: '上传记录不存在' });
    }

    const uploadedChunks = metaManager.getUploadedChunks(fileId);

    res.json({
        code: 0,
        data: {
            ...meta,
            uploadedChunks,
            progress: Math.round((uploadedChunks.length / meta.totalChunks) * 100)
        }
    });
});

app.use('/uploads', express.static(UPLOAD_DIR));

app.listen(3001, () => {
    console.log('断点续传服务启动: http://localhost:3001');
});
```

## 5. 方案三：流式上传

### 5.1 原理

利用 `ReadableStream` 和 `fetch` 的流式 API，将文件以流的形式逐块读取并发送到服务端，服务端通过 `WritableStream` 逐块写入磁盘。不需要在内存中缓存整个文件或分片。

```
┌─────────────────────────────────┐    ┌──────────────────────────────┐
│         客户端 (Browser)         │    │        服务端 (Node.js)        │
│                                 │    │                              │
│  file.stream()                  │    │  app.post('/api/upload/stream')
│  ┌───────────┐ 管道式传输        │    │  ┌──────────────────────────┐ │
│  │ Chunk 1   │ ────────────────► │    │  │  req.pipe(writeStream)   │ │
│  │ Chunk 2   │ ────────────────► │    │  │                          │ │
│  │ Chunk 3   │ ────────────────► │    │  │  fs.createWriteStream()  │ │
│  │ ...       │ ────────────────► │    │  │                          │ │
│  │ Chunk N   │ ────────────────► │    │  └──────────────────────────┘ │
│  └───────────┘                   │    │                              │
│                                 │    │  内存占用: 极小（流式处理）     │
│  内存占用: 极小（流式读取）       │    │                              │
└─────────────────────────────────┘    └──────────────────────────────┘
```

### 5.2 前端实现

```javascript
/**
 * 方案三：流式上传器
 *
 * 特性：
 *  - 使用 ReadableStream 流式读取文件
 *  - 使用 fetch 流式发送
 *  - 实时进度（通过自定义进度回调）
 *  - 支持暂停/取消
 *  - 极低内存占用（适合 GB 级文件）
 *
 * 注意：需要浏览器支持 ReadableStream 和 pipeline API
 * 兼容性：Chrome 78+, Edge 79+, Firefox 100+, Safari 15.4+
 */
class StreamingUploader {
    /**
     * @param {Object} options
     * @param {File}   options.file
     * @param {string} options.uploadUrl
     * @param {number} [options.chunkSize=64*1024]  - 流式读取块大小（默认 64KB）
     * @param {Function} [options.onProgress]
     * @param {Function} [options.onSuccess]
     * @param {Function} [options.onError]
     */
    constructor(options) {
        this.file = options.file;
        this.uploadUrl = options.uploadUrl;
        this.chunkSize = options.chunkSize || 64 * 1024; // 64KB

        this.onProgress = options.onProgress || (() => {});
        this.onSuccess = options.onSuccess || (() => {});
        this.onError = options.onError || (() => {});

        this.status = UploadStatus.PENDING;
        this.abortController = null;
        this.uploadedBytes = 0;
    }

    /**
     * 创建带进度跟踪的 ReadableStream
     */
    createProgressStream() {
        const file = this.file;
        const chunkSize = this.chunkSize;
        let offset = 0;
        const self = this;

        return new ReadableStream({
            async pull(controller) {
                if (self.status === UploadStatus.PAUSED) {
                    // 暂停：等待恢复
                    await new Promise(resolve => {
                        const check = setInterval(() => {
                            if (self.status === UploadStatus.UPLOADING) {
                                clearInterval(check);
                                resolve();
                            }
                        }, 100);
                    });
                }

                if (offset >= file.size) {
                    controller.close();
                    return;
                }

                const end = Math.min(offset + chunkSize, file.size);
                const blob = file.slice(offset, end);

                // 读取分片
                const arrayBuffer = await blob.arrayBuffer();
                controller.enqueue(new Uint8Array(arrayBuffer));

                offset = end;
                self.uploadedBytes = offset;

                // 进度回调
                const progress = Math.round((offset / file.size) * 100);
                self.onProgress(progress, offset, file.size);
            }
        });
    }

    /**
     * 开始上传
     */
    async start() {
        if (this.status === UploadStatus.UPLOADING) return;

        this.status = UploadStatus.UPLOADING;
        this.abortController = new AbortController();
        this.uploadedBytes = 0;

        try {
            const stream = this.createProgressStream();

            const response = await fetch(this.uploadUrl, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/octet-stream',
                    'X-File-Name': encodeURIComponent(this.file.name),
                    'X-File-Size': String(this.file.size),
                    'X-File-Type': this.file.type || 'application/octet-stream'
                },
                body: stream,
                duplex: 'half', // 必需：允许在请求发送时读取响应
                signal: this.abortController.signal
            });

            if (!response.ok) {
                throw new Error(`上传失败: ${response.status}`);
            }

            const result = await response.json();
            this.status = UploadStatus.COMPLETED;
            this.onSuccess(result);
        } catch (error) {
            if (error.name !== 'AbortError') {
                this.status = UploadStatus.FAILED;
                this.onError(error);
            }
        }
    }

    pause() {
        this.status = UploadStatus.PAUSED;
    }

    resume() {
        this.status = UploadStatus.UPLOADING;
    }

    cancel() {
        this.status = UploadStatus.FAILED;
        if (this.abortController) {
            this.abortController.abort();
        }
    }
}

// ==================== 辅助：非流式分片上传（兼容方案） ====================
/**
 * 兼容方案：使用分片循环 + fetch 实现类似流式上传
 * 适用于不支持 ReadableStream 的浏览器
 */
class FallbackStreamingUploader {
    constructor(options) {
        this.file = options.file;
        this.uploadUrl = options.uploadUrl;
        this.chunkSize = options.chunkSize || 1024 * 1024; // 1MB

        this.onProgress = options.onProgress || (() => {});
        this.onSuccess = options.onSuccess || (() => {});
        this.onError = options.onError || (() => {});

        this.status = UploadStatus.PENDING;
        this.abortController = null;
        this.uploadedBytes = 0;
    }

    async start() {
        this.status = UploadStatus.UPLOADING;
        this.abortController = new AbortController();
        this.uploadedBytes = 0;

        try {
            const totalChunks = Math.ceil(this.file.size / this.chunkSize);

            for (let i = 0; i < totalChunks; i++) {
                if (this.status === UploadStatus.PAUSED) {
                    await this.waitForResume();
                }

                const start = i * this.chunkSize;
                const end = Math.min(start + this.chunkSize, this.file.size);
                const blob = this.file.slice(start, end);

                const formData = new FormData();
                formData.append('chunk', blob);
                formData.append('index', String(i));
                formData.append('fileName', this.file.name);
                formData.append('totalChunks', String(totalChunks));

                const response = await fetch(this.uploadUrl, {
                    method: 'POST',
                    body: formData,
                    signal: this.abortController.signal
                });

                if (!response.ok) {
                    throw new Error(`分片 ${i} 上传失败: ${response.status}`);
                }

                this.uploadedBytes = end;
                const progress = Math.round((end / this.file.size) * 100);
                this.onProgress(progress, end, this.file.size);
            }

            this.status = UploadStatus.COMPLETED;
            this.onSuccess({ message: '上传完成' });
        } catch (error) {
            if (error.name !== 'AbortError') {
                this.status = UploadStatus.FAILED;
                this.onError(error);
            }
        }
    }

    waitForResume() {
        return new Promise(resolve => {
            const check = setInterval(() => {
                if (this.status === UploadStatus.UPLOADING) {
                    clearInterval(check);
                    resolve();
                }
            }, 100);
        });
    }

    pause() { this.status = UploadStatus.PAUSED; }
    resume() { this.status = UploadStatus.UPLOADING; }
    cancel() {
        this.status = UploadStatus.FAILED;
        if (this.abortController) this.abortController.abort();
    }
}
```

## 6. 文件上传进度监听与展示

> 文件上传进度监听是提升用户体验的关键环节，尤其在大文件上传场景下，实时、准确的进度反馈能有效缓解用户焦虑、避免重复操作。本节系统介绍前端实现上传进度监听的两种核心方案、进度计算方式、实时展示方案、跨域问题处理及兼容性方案。

### 6.2 方案一：XMLHttpRequest 的 progress 事件

#### 6.2.1 核心原理

`XMLHttpRequest` 提供了 `upload` 属性，这是一个 `XMLHttpRequestUpload` 对象，上面可以监听 `progress` 事件。浏览器在上传过程中会周期性触发该事件，事件对象中包含已上传字节数和总字节数。

#### 6.2.2 完整代码示例

```javascript
/**
 * 基于 XMLHttpRequest 的带进度上传
 *
 * @param {Object}   options
 * @param {string}   options.url           - 上传接口地址
 * @param {FormData|Blob|string} options.body - 请求体
 * @param {Function} options.onProgress    - 进度回调 (loaded, total, percent)
 * @param {Function} options.onSuccess     - 成功回调 (response)
 * @param {Function} options.onError       - 错误回调 (error)
 * @param {AbortSignal} [options.signal]   - 取消信号（用于暂停/取消）
 * @param {Object}   [options.headers]     - 自定义请求头
 * @returns {XMLHttpRequest} xhr 实例（用于进一步控制）
 */
function uploadWithProgress(options) {
    const {
        url,
        body,
        onProgress = () => {},
        onSuccess = () => {},
        onError = () => {},
        signal,
        headers = {}
    } = options;

    const xhr = new XMLHttpRequest();

    // ★ 关键1：监听 upload 的 progress 事件
    xhr.upload.addEventListener('progress', (event) => {
        // event.lengthComputable: 是否可以计算总长度
        // event.loaded: 已上传的字节数
        // event.total: 需要上传的总字节数
        if (event.lengthComputable) {
            const percent = Math.round((event.loaded / event.total) * 100);
            onProgress(event.loaded, event.total, percent);
        }
    });

    // ★ 关键2：监听上传完成
    xhr.upload.addEventListener('load', () => {
        console.log('上传完成，等待服务端响应...');
    });

    // ★ 关键3：监听上传错误
    xhr.upload.addEventListener('error', () => {
        onError(new Error('上传过程中发生网络错误'));
    });

    // ★ 关键4：监听上传中止
    xhr.upload.addEventListener('abort', () => {
        console.log('上传已被中止');
    });

    // ★ 关键5：监听服务端响应
    xhr.addEventListener('load', () => {
        if (xhr.status >= 200 && xhr.status < 300) {
            let result;
            try {
                result = JSON.parse(xhr.responseText);
            } catch {
                result = xhr.responseText;
            }
            onSuccess(result);
        } else {
            onError(new Error(`服务端返回错误: ${xhr.status}`));
        }
    });

    // ★ 关键6：监听网络错误
    xhr.addEventListener('error', () => {
        onError(new Error('网络请求失败'));
    });

    // ★ 关键7：监听超时
    xhr.addEventListener('timeout', () => {
        onError(new Error('请求超时'));
    });

    // 初始化请求
    xhr.open('POST', url);

    // 设置超时（默认 0 表示不超时，大文件建议设置较长超时）
    xhr.timeout = 10 * 60 * 1000; // 10 分钟

    // 设置自定义请求头
    Object.entries(headers).forEach(([key, value]) => {
        xhr.setRequestHeader(key, value);
    });

    // 处理取消信号
    if (signal) {
        if (signal.aborted) {
            xhr.abort();
        } else {
            signal.addEventListener('abort', () => xhr.abort());
        }
    }

    // 发送请求
    xhr.send(body);

    return xhr; // 返回实例以便外部调用 xhr.abort()
}

// ==================== 使用示例 ====================

// 示例1：上传单个文件
function uploadSingleFile() {
    const fileInput = document.getElementById('fileInput');
    const file = fileInput.files[0];
    if (!file) return;

    const formData = new FormData();
    formData.append('file', file);
    formData.append('userId', '123');

        url: '/api/upload',

            // 更新进度条
            document.getElementById('progressBar').style.width = percent + '%';
            document.getElementById('progressText').textContent =
                `${percent}% (${formatSize(loaded)} / ${formatSize(total)})`;

            // 计算上传速度和剩余时间
            const elapsed = (Date.now() - startTime) / 1000; // 秒
            const speed = loaded / elapsed; // 字节/秒
            const remaining = (total - loaded) / speed; // 剩余秒数

            console.log(`速度: ${formatSize(speed)}/s, 剩余: ${formatTime(remaining)}`);
        },
        onSuccess: (result) => {
            console.log('上传成功:', result);
        },
        onError: (error) => {
            console.error('上传失败:', error);
        }
    });
}

    if (seconds < 60) return Math.ceil(seconds) + '秒';

    return (seconds / 3600).toFixed(1) + '小时';
```

#### 6.2.3 progress 事件关键参数说明

| 参数 | 类型 | 说明 |
|------|------|------|
| `event.lengthComputable` | `boolean` | 是否可计算总长度。`true` 时 `loaded/total` 才有意义。如果服务端使用 chunked 编码且未设置 Content-Length，可能为 `false` |
| `event.loaded` | `number` | 已上传的字节数（含请求头 + 请求体） |
| `event.total` | `number` | 总字节数（含请求头 + 请求体） |
| `event.timeStamp` | `DOMHighResTimeStamp` | 事件触发的时间戳 |

> **注意**：`loaded` 和 `total` 包含的是**整个请求体**的大小，如果使用 `FormData`，除了文件本身还包括其他字段的字节和分隔符等开销，因此 `total` 可能略大于文件实际大小。

#### 6.2.4 事件触发频率

`progress` 事件并非每字节都触发，浏览器会根据上传速率周期性触发（通常每 50ms~250ms 一次）。对于极小文件，可能只触发一次甚至不触发（上传太快）。

```javascript
// 监听 progress 事件的触发频率
let triggerCount = 0;
let lastTime = Date.now();

xhr.upload.addEventListener('progress', (e) => {
    triggerCount++;
    const now = Date.now();
    const interval = now - lastTime;
    lastTime = now;
    console.log(`第 ${triggerCount} 次触发, 间隔 ${interval}ms, 进度 ${Math.round(e.loaded / e.total * 100)}%`);
});
```

### 6.3 方案二：Fetch API + ReadableStream

#### 6.3.1 核心原理

`fetch` API 本身不提供上传进度事件，但可以通过将 `body` 设置为自定义的 `ReadableStream`，在流的 `pull` 函数中追踪已发送的字节数，从而实现进度监听。

#### 6.3.2 完整代码示例

```javascript
/**
 * 基于 Fetch + ReadableStream 的带进度上传
 *
 * 原理：自定义 ReadableStream，在每次 pull（读取数据块）时
 *       累加已发送字节数，调用进度回调
 *
 * @param {Object}   options
 * @param {string}   options.url         - 上传接口地址
 * @param {File|Blob} options.file       - 要上传的文件
 * @param {number}   [options.chunkSize=65536] - 流式读取块大小（默认 64KB）
 * @param {Function} options.onProgress  - 进度回调 (loaded, total, percent)
 * @param {Function} options.onSuccess   - 成功回调 (response)
 * @param {Function} options.onError     - 错误回调 (error)
 * @param {AbortSignal} [options.signal] - 取消信号
 * @param {Object}   [options.headers]   - 自定义请求头
 * @returns {Promise<Response>}
 */
async function uploadWithFetchProgress(options) {
    const {
        url,
        file,
        chunkSize = 64 * 1024, // 64KB
        onProgress = () => {},
        onSuccess = () => {},
        onError = () => {},
        signal,
        headers = {}
    } = options;

    const totalBytes = file.size;
    let uploadedBytes = 0;

    /**
     * 创建带进度追踪的 ReadableStream
     *
     * pull 函数在流需要更多数据时被调用
     * 我们在这里读取文件的下一块数据并更新进度
     */
    const progressStream = new ReadableStream({
        async pull(controller) {
            // 检查是否被取消
            if (signal?.aborted) {
                controller.error(new DOMException('上传被取消', 'AbortError'));
                return;
            }

            // 读取下一块数据
            const blob = file.slice(uploadedBytes, uploadedBytes + chunkSize);
            const buffer = await blob.arrayBuffer();

            if (buffer.byteLength === 0) {
                // 数据读取完毕，关闭流
                controller.close();
                return;
            }

            // 将数据推入流中（fetch 会发送这些数据）
            controller.enqueue(new Uint8Array(buffer));

            // ★ 更新进度
            uploadedBytes += buffer.byteLength;
            const percent = Math.round((uploadedBytes / totalBytes) * 100);
            onProgress(uploadedBytes, totalBytes, percent);

            // 如果已读取完毕，关闭流
            if (uploadedBytes >= totalBytes) {
                controller.close();
            }
        },

        // cancel 函数在流被取消时调用
        cancel(reason) {
            console.log('上传流被取消:', reason);
        }
    });

    try {
        const response = await fetch(url, {
            method: 'POST',
            headers: {
                'Content-Type': file.type || 'application/octet-stream',
                'Content-Length': String(totalBytes),
                'X-File-Name': encodeURIComponent(file.name),
                ...headers
            },
            body: progressStream,
            duplex: 'half', // ★ 必需：允许请求体为流式
            signal
        });

        if (!response.ok) {
            throw new Error(`上传失败: ${response.status}`);
        }

        const result = await response.json();
        onProgress(totalBytes, totalBytes, 100); // 确保进度到 100%
        onSuccess(result);
        return result;
    } catch (error) {
        if (error.name !== 'AbortError') {
            onError(error);
        }
        throw error;
    }
}

// ==================== 使用示例 ====================

async function uploadWithFetch() {
    const file = document.getElementById('fileInput').files[0];
    if (!file) return;

    const controller = new AbortController();
    const startTime = Date.now();

    await uploadWithFetchProgress({
        url: '/api/upload/stream',
        file,
        chunkSize: 256 * 1024, // 256KB

        },
        onSuccess: (result) => {
            console.log('上传成功:', result);
        },
        onError: (error) => {
            console.error('上传失败:', error);
        }
    });

    // 取消上传
    // document.getElementById('cancelBtn').onclick = () => controller.abort();
}
```

#### 6.3.3 Fetch 进度方案的局限性

```
┌─────────────────────────────────────────────────────────────────────────┐
│                    Fetch 进度方案的局限性                                 │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│  ⚠️ 局限1：只能追踪"发送进度"，不能追踪"服务端接收进度"                  │
│  → ReadableStream 的 pull 追踪的是"数据被读取并发送"的进度             │
│  → 但数据可能还在浏览器缓冲区中，尚未真正通过网络发送出去               │
│  → 实际发送进度可能滞后于 pull 进度                                     │
│                                                                         │
│  ⚠️ 局限2：浏览器兼容性                                                 │
│  → ReadableStream 作为 fetch body：Chrome 78+, Edge 79+               │
│  → duplex: 'half' 选项：Chrome 105+                                    │
│  → Firefox/Safari 支持较晚或有限                                        │
│                                                                         │
│  ⚠️ 局限3：无法获取精确的网络层进度                                     │
│  → XHR 的 progress 事件来自浏览器网络层，更接近真实传输进度              │
│  → Fetch 的 ReadableStream 方案只能追踪应用层的数据读取进度             │
│                                                                         │
│  ⚠️ 局限4：Content-Length 必须手动设置                                  │
│  → 流式 body 时浏览器无法自动计算 Content-Length                       │
│  → 需要手动在 headers 中设置，否则服务端可能无法预知文件大小            │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘
```

### 6.4 两种方案对比

| 维度 | XHR + progress | Fetch + ReadableStream |
|------|---------------|----------------------|
| **兼容性** | IE10+，全浏览器支持 | Chrome 105+，较新浏览器 |
| **进度精确度** | 高（网络层进度） | 中（应用层读取进度） |
| **API 风格** | 回调 | Promise + 流式 |
| **取消方式** | `xhr.abort()` | `AbortController.abort()` |
| **上传速度** | 与 XHR 性能一致 | 与 Fetch 性能一致 |
| **文件类型支持** | FormData / Blob / ArrayBuffer | ReadableStream / Blob |
| **服务端接收进度** | ✅ 可获取（upload.progress） | ❌ 无法获取 |
| **超时控制** | `xhr.timeout` | 需用 AbortController 模拟 |
| **推荐场景** | 需要精确进度、兼容旧浏览器 | 现代项目、流式上传 |

### 6.5 进度计算方式

#### 6.5.1 基本计算公式

```javascript
// 进度百分比 = 已上传字节数 / 总字节数 × 100
const percent = (loaded / total) * 100;

// 示例
const loaded = 5242880; // 5MB
const total  = 10485760; // 10MB
const percent = (5242880 / 10485760) * 100; // 50%
```

#### 6.5.2 分片上传的进度计算

```javascript
/**
 * 分片上传场景的进度计算
 *
 * 注意：分片上传时，每个分片是独立请求
 * 进度 = 已完成分片数 / 总分片数 × 100
 * 或更精确：已完成分片总字节 / 文件总字节 × 100
 */
class ChunkProgressTracker {
    constructor(fileSize, totalChunks) {
        this.fileSize = fileSize;
        this.totalChunks = totalChunks;
        this.completedBytes = 0;
        this.completedChunks = new Set();
        this.currentChunkProgress = new Map(); // index → loaded bytes
    }

    /**
     * 更新单个分片的上传进度
     * @param {number} chunkIndex - 分片索引
     * @param {number} loaded     - 该分片已上传字节数
     * @param {number} total      - 该分片总字节数
     */
    updateChunkProgress(chunkIndex, loaded, total) {
        this.currentChunkProgress.set(chunkIndex, { loaded, total });
        this.recalculate();
    }

    /**
     * 标记某个分片已完成
     * @param {number} chunkIndex - 分片索引
     * @param {number} chunkSize  - 该分片大小
     */
    markChunkComplete(chunkIndex, chunkSize) {
        this.completedChunks.add(chunkIndex);
        this.completedBytes += chunkSize;
        this.currentChunkProgress.delete(chunkIndex);
        this.recalculate();
    }

    /**
     * 重新计算总进度
     */
    recalculate() {
        // 方式1：基于已完成分片数（粗略）
        const chunkBasedPercent = (this.completedChunks.size / this.totalChunks) * 100;

        // 方式2：基于字节数（精确）
        let inProgressBytes = 0;
        this.currentChunkProgress.forEach(({ loaded }) => {
            inProgressBytes += loaded;
        });
        const totalUploaded = this.completedBytes + inProgressBytes;
        const byteBasedPercent = (totalUploaded / this.fileSize) * 100;

        // 使用字节级进度（更精确）
        this.percent = Math.min(100, Math.round(byteBasedPercent));
        this.loadedBytes = totalUploaded;
    }

    getProgress() {
        return {
            percent: this.percent,
            loaded: this.loadedBytes,
            total: this.fileSize
        };
    }
}

// 使用示例
const tracker = new ChunkProgressTracker(100 * 1024 * 1024, 20); // 100MB, 20分片

// 某个分片正在上传中
tracker.updateChunkProgress(0, 2500000, 5000000); // 分片0上传了50%

// 分片0完成
tracker.markChunkComplete(0, 5000000);

console.log(tracker.getProgress());
// { percent: 5, loaded: 5000000, total: 104857600 }
```

#### 6.5.3 上传速度与剩余时间计算

```javascript
/**
 * 上传速度和剩余时间计算器
 */
class SpeedCalculator {
    constructor() {
        this.samples = [];      // 采样点：{ time, loaded }
        this.maxSamples = 10;   // 最大采样数
    }

    /**
     * 记录采样点
     * @param {number} loaded - 已上传字节数
     */
    record(loaded) {
        const now = Date.now();
        this.samples.push({ time: now, loaded });

        // 保留最近的 N 个采样点
        if (this.samples.length > this.maxSamples) {
            this.samples.shift();
        }
    }

    /**
     * 计算上传速度（字节/秒）
     */
    getSpeed() {
        if (this.samples.length < 2) return 0;

        const first = this.samples[0];
        const last = this.samples[this.samples.length - 1];
        const timeDiff = (last.time - first.time) / 1000; // 秒
        const bytesDiff = last.loaded - first.loaded;

        if (timeDiff <= 0) return 0;
        return bytesDiff / timeDiff; // 字节/秒
    }

    /**
     * 计算剩余时间（秒）
     * @param {number} remainingBytes - 剩余字节数
     */
    getRemainingTime(remainingBytes) {
        const speed = this.getSpeed();
        if (speed <= 0) return Infinity;
        return remainingBytes / speed;
    }

    /**
     * 重置
     */
    reset() {
        this.samples = [];
    }
}

// 使用示例
const speedCalc = new SpeedCalculator();

xhr.upload.addEventListener('progress', (e) => {
    speedCalc.record(e.loaded);

    const speed = speedCalc.getSpeed();
    const remainingBytes = e.total - e.loaded;
    const remainingTime = speedCalc.getRemainingTime(remainingBytes);

    console.log(`进度: ${Math.round(e.loaded / e.total * 100)}%`);
    console.log(`速度: ${formatSize(speed)}/s`);
    console.log(`剩余: ${formatTime(remainingTime)}`);
});
```

```javascript
/**
 * 进度 UI 控制器
 * 使用 requestAnimationFrame 优化渲染性能
 */
class ProgressUI {
    constructor() {
        this.elements = {
            bar: document.getElementById('progressFill'),
            percent: document.getElementById('progressPercent'),
            uploadedSize: document.getElementById('uploadedSize'),
            totalSize: document.getElementById('totalSize'),
            speed: document.getElementById('uploadSpeed'),
            remaining: document.getElementById('remainingTime'),
        };
        this.pendingUpdate = null;
        this.rafId = null;
    }

    /**
     * 更新进度显示
     * 使用 requestAnimationFrame 合并高频更新，避免渲染抖动
     */
    update(data) {
        this.pendingUpdate = data;

        if (!this.rafId) {
            this.rafId = requestAnimationFrame(() => {
                this.render(this.pendingUpdate);
                this.rafId = null;
            });
        }
    }

    render(data) {
        const { percent, loaded, total, speed, remainingTime } = data;

        this.elements.bar.style.width = percent + '%';
        this.elements.percent.textContent = percent + '%';
        this.elements.uploadedSize.textContent = formatSize(loaded);
        this.elements.totalSize.textContent = formatSize(total);
        this.elements.speed.textContent = formatSize(speed) + '/s';
        this.elements.remaining.textContent = formatTime(remainingTime);
    }

    reset() {
        if (this.rafId) {
            cancelAnimationFrame(this.rafId);
            this.rafId = null;
        }
        this.render({ percent: 0, loaded: 0, total: 0, speed: 0, remainingTime: 0 });
    }
}

// 使用示例
```

```javascript
app.use(cors({
    origin: (origin, callback) => {
        // 允许的域名白名单

            'https://app.example.com',
            'https://staging.example.com',
            'http://localhost:8080'  // 开发环境

        if (!origin || allowedOrigins.includes(origin)) {

    maxAge: 86400  // 预检请求缓存时间（秒）

    res.header('Access-Control-Expose-Headers', 'Content-Length');

    // 处理预检请求

        return res.sendStatus(204);
    }
```

```javascript
/**
```

```javascript
function createUploader(options) {
    // 检测是否支持 ReadableStream 作为 fetch body
    const supportsFetchStream = (() => {
        try {
            const stream = new ReadableStream({});
            const req = new Request('', {
                method: 'POST',
                body: stream,
                duplex: 'half'
            });
            return req.body === stream;
        } catch {
            return false;
        }
    })();

    // 检测是否支持 XMLHttpRequest
    const supportsXHR = typeof XMLHttpRequest !== 'undefined';

    if (supportsFetchStream && options.preferFetch !== false) {
        console.log('使用 Fetch + ReadableStream 方案');
        return uploadWithFetchProgress(options);
    } else if (supportsXHR) {
        console.log('降级使用 XHR + progress 方案');
        return uploadWithProgress(options);
    } else {
        // 最终降级：无进度反馈的普通上传
        console.warn('当前环境不支持上传进度监听，使用普通上传');
        return uploadWithoutProgress(options);
    }
}

 * 无进度反馈的降级上传方案
 */
async function uploadWithoutProgress(options) {
    const formData = new FormData();
    formData.append('file', options.file);

    try {
        const response = await fetch(options.url, {
            method: 'POST',
            body: formData
        });
        const result = await response.json();
        options.onSuccess?.(result);
    } catch (error) {
        options.onError?.(error);
    }
}
```

### 7.2 上传速度对比（理论分析）

```
场景：上传 1GB 文件，网络带宽 100Mbps

分片上传（并发 3）：
  分片大小 5MB × 200 个分片
  实际并发 3 → 每批 3 个分片 = 15MB
  理论耗时 ≈ 1024MB / (100Mbps/8) ≈ 82 秒
  实际耗时 ≈ 90~120 秒（含合并开销）

断点续传（并发 3）：
  首次上传：同分片上传
  断网重传：仅上传缺失分片，假设已传 80%
  重传耗时 ≈ 82 × 20% ≈ 16 秒 + Hash 计算时间

流式上传：
  单连接顺序传输
  理论耗时 ≈ 1024MB / (100Mbps/8) ≈ 82 秒
  实际耗时 ≈ 85~100 秒（无并发优势，但无合并开销）
```

### 7.3 优缺点详细分析

#### 分片上传

**优点：**
- 实现简单，成熟稳定，社区方案丰富
- 并发上传可充分利用带宽
- 浏览器兼容性非常好
- 单个分片失败只影响该分片，可独立重试
- 适合中等大小文件（100MB~500MB）

**缺点：**
- 断网后无法恢复，需重新上传全部分片
- 服务端需要额外存储分片和合并操作
- 磁盘 I/O 较高（先写分片，再合并）
- 需要两次请求（上传 + 合并）

#### 断点续传

**优点：**
- 断网后可恢复，用户体验最佳
- 文件 hash 去重，相同文件秒传
- 支持页面刷新/关闭后恢复
- 适合超大文件（500MB~10GB+）
- 并发上传，速度最优

**缺点：**
- 实现复杂度最高
- 客户端计算 hash 耗时（大文件可能需要数秒到数十秒）
- 需要 SparkMD5 等额外依赖
- 服务端需要维护上传状态和分片索引
- 处理不当可能导致磁盘空间浪费（未清理的分片）

#### 流式上传

**优点：**
- 内存占用极低（客户端和服务端均为流式处理）
- 服务端无需合并操作，减少磁盘 I/O
- 实时进度反馈最精确（字节级）
- 实现相对简洁
- 适合服务器端资源受限的场景

**缺点：**
- 浏览器兼容性较差（需要 ReadableStream 支持）
- 单连接，无法并发，大文件上传速度受限
- 断网后需从头开始
- 暂停/恢复实现复杂（需自定义流控制）
- 无法在上传失败时部分重传

```javascript
function getOptimalChunkSize(fileSize) {
    if (fileSize < 10 * 1024 * 1024)  return 1 * 1024 * 1024;   // <10MB: 1MB
    if (fileSize < 100 * 1024 * 1024) return 2 * 1024 * 1024;   // <100MB: 2MB
    if (fileSize < 1024 * 1024 * 1024) return 5 * 1024 * 1024;  // <1GB: 5MB
    return 10 * 1024 * 1024;                                      // >1GB: 10MB
```

#### 2. 并发数动态调整

```javascript
/**
 * 根据网络状况动态调整并发数
 */
class AdaptiveConcurrency {
    constructor(initial = 3) {
        this.concurrency = initial;
        this.successCount = 0;
        this.failCount = 0;
        this.latencies = [];
    }

    recordSuccess(latency) {
        this.successCount++;
        this.latencies.push(latency);
        if (this.latencies.length > 10) this.latencies.shift();

        // 连续成功且延迟低 → 增加并发
        if (this.successCount >= 5 && this.getAvgLatency() < 500) {
            this.concurrency = Math.min(6, this.concurrency + 1);
            this.successCount = 0;
        }
    }

    recordFailure() {
        this.failCount++;
        // 连续失败 → 降低并发
        if (this.failCount >= 3) {
            this.concurrency = Math.max(1, this.concurrency - 1);
            this.failCount = 0;
        }
    }

    getAvgLatency() {
        if (this.latencies.length === 0) return 0;
        return this.latencies.reduce((a, b) => a + b, 0) / this.latencies.length;
    }
}
```

#### 3. Web Worker 计算 Hash

```javascript
// hash-worker.js — 在 Worker 中计算 MD5，避免阻塞主线程
self.onmessage = async function(e) {
    const { file } = e.data;
    const spark = new SparkMD5.ArrayBuffer();
    const chunkSize = 5 * 1024 * 1024; // 5MB

    let offset = 0;
    while (offset < file.size) {
        const end = Math.min(offset + chunkSize, file.size);
        const blob = file.slice(offset, end);
        const buffer = await blob.arrayBuffer();
        spark.append(buffer);

        offset = end;
        const progress = Math.round((offset / file.size) * 100);
        self.postMessage({ type: 'progress', progress });
    }

    const hash = spark.end();
    self.postMessage({ type: 'complete', hash });
};

// 主线程使用
function computeHashInWorker(file) {
    return new Promise((resolve, reject) => {
        const worker = new Worker('hash-worker.js');
        worker.onmessage = (e) => {
            if (e.data.type === 'progress') {
                console.log(`Hash 计算进度: ${e.data.progress}%`);
            } else if (e.data.type === 'complete') {
                worker.terminate();
                resolve(e.data.hash);
            }
        };
        worker.onerror = reject;
        worker.postMessage({ file });
    });
}
```
