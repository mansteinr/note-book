## 第一部分：前端高并发解决方案

> **问题场景**：高并发场景下，前端面临大量用户同时访问、高频请求触发、大量 DOM 渲染、计算密集型任务阻塞主线程等问题，导致页面卡顿甚至崩溃。

### 1.1 请求防抖与节流

#### 原理

防抖（Debounce）和节流（Throttle）是控制函数执行频率的核心手段，用于限制高频触发的事件（如 scroll、resize、input）中回调的执行次数。

- **防抖**：事件触发后等待 N 毫秒再执行，若期间再次触发则重新计时。适用于输入搜索、表单验证等场景。
- **节流**：在 N 毫秒内只执行一次，无论触发多少次。适用于滚动监听、拖拽跟随等场景。

#### 实现代码

```javascript
/**
 * 防抖函数
 * @param {Function} fn - 需要防抖的函数
 * @param {number} delay - 延迟时间(ms)
 * @param {boolean} immediate - 是否立即执行
 */
function debounce(fn, delay = 300, immediate = false) {
  let timer = null;

  return function (...args) {
    const context = this;

    if (timer) clearTimeout(timer);

    if (immediate && !timer) {
      fn.apply(context, args);
    }

    timer = setTimeout(() => {
      if (!immediate) {
        fn.apply(context, args);
      }
      timer = null;
    }, delay);
  };
}

/**
 * 节流函数
 * @param {Function} fn - 需要节流的函数
 * @param {number} interval - 间隔时间(ms)
 * @param {object} options - { leading: 是否首次立即执行, trailing: 是否尾随执行 }
 */
function throttle(fn, interval = 300, options = { leading: true, trailing: true }) {
  let timer = null;
  let lastTime = 0;

  return function (...args) {
    const context = this;
    const now = Date.now();

    if (!lastTime && !options.leading) {
      lastTime = now;
    }

    const remaining = interval - (now - lastTime);

    if (remaining <= 0 || remaining > interval) {
      if (timer) {
        clearTimeout(timer);
        timer = null;
      }
      lastTime = now;
      fn.apply(context, args);
    } else if (!timer && options.trailing) {
      timer = setTimeout(() => {
        lastTime = options.leading ? Date.now() : 0;
        timer = null;
        fn.apply(context, args);
      }, remaining);
    }
  };
}
```

#### 使用示例

```javascript
// 搜索输入防抖
const searchInput = document.getElementById('search');
searchInput.addEventListener('input', debounce(function (e) {
  console.log('发起搜索请求:', e.target.value);
  fetchSearchResults(e.target.value);
}, 500));

// 滚动事件节流
window.addEventListener('scroll', throttle(function () {
  console.log('滚动位置:', window.scrollY);
  updateScrollIndicator();
}, 200));
```

#### 高并发场景适用性

- **搜索联想**：大量用户同时输入搜索关键词，防抖可减少 80% 以上的无效请求
- **窗口 resize 重绘**：节流避免高频率的重排重绘
- **按钮连续点击**：防止用户快速点击导致重复提交

### 1.2 请求队列与并发控制

#### 原理

浏览器对同域名的并发请求有限制（HTTP/1.1 约 6 个），当同时发起大量请求时，需要队列化管理，控制并发数量，避免队列溢出和带宽竞争。

#### 实现代码

```javascript
/**
 * 并发请求控制器
 */
class ConcurrencyController {
  /**
   * @param {number} limit - 最大并发数
   */
  constructor(limit = 6) {
    this.limit = limit;
    this.running = 0;
    this.queue = [];
  }

  /**
   * 添加任务到队列
   * @param {Function} taskFn - 返回 Promise 的任务函数
   * @returns {Promise}
   */
  add(taskFn) {
    return new Promise((resolve, reject) => {
      this.queue.push({ taskFn, resolve, reject });
      this._run();
    });
  }

  _run() {
    while (this.running < this.limit && this.queue.length > 0) {
      const { taskFn, resolve, reject } = this.queue.shift();
      this.running++;

      taskFn()
        .then(resolve)
        .catch(reject)
        .finally(() => {
          this.running--;
          this._run();
        });
    }
  }
}

// --- 使用示例 ---
const controller = new ConcurrencyController(4);

const urls = Array.from({ length: 50 }, (_, i) => `/api/data/${i}`);

const promises = urls.map((url) =>
  controller.add(() => fetch(url).then((res) => res.json()))
);

Promise.all(promises).then((results) => {
  console.log('所有请求完成，共', results.length, '条数据');
});
```

#### 请求优先级队列

```javascript
/**
 * 带优先级的请求队列
 */
class PriorityRequestQueue {
  constructor(limit = 6) {
    this.limit = limit;
    this.running = 0;
    this.queues = { high: [], normal: [], low: [] };
  }

  add(taskFn, priority = 'normal') {
    return new Promise((resolve, reject) => {
      this.queues[priority].push({ taskFn, resolve, reject });
      this._run();
    });
  }

  _getNextTask() {
    for (const level of ['high', 'normal', 'low']) {
      if (this.queues[level].length > 0) {
        return this.queues[level].shift();
      }
    }
    return null;
  }

  _run() {
    while (this.running < this.limit) {
      const task = this._getNextTask();
      if (!task) break;

      this.running++;
      const { taskFn, resolve, reject } = task;

      taskFn()
        .then(resolve)
        .catch(reject)
        .finally(() => {
          this.running--;
          this._run();
        });
    }
  }
}
```

#### 高并发场景适用性

- **批量上传/下载**：控制并发数避免浏览器连接池耗尽
- **数据看板**：大量图表数据请求按优先级加载
- **分页预加载**：控制预加载请求数量，避免带宽竞争

### 1.3 Web Worker 多线程处理

#### 原理

JavaScript 是单线程语言，复杂计算会阻塞主线程导致页面卡顿。Web Worker 允许在后台线程中执行脚本，不干扰主线程的 UI 渲染和用户交互。

```
┌────────────────────────────────────────────────────────────┐
│                      主线程 (Main Thread)                    │
│  ┌──────────┐  ┌──────────┐  ┌──────────┐                  │
│  │ UI 渲染  │  │ 事件处理  │  │ 轻量计算  │                  │
│  └──────────┘  └──────────┘  └──────────┘                  │
│       │              │                                        │
│       │  postMessage │  onmessage                            │
│       ▼              ▼                                        │
│  ┌─────────────────────────────────────────────────────┐    │
│  │              Web Worker 线程                          │    │
│  │  ┌──────────┐  ┌──────────┐  ┌──────────┐           │    │
│  │  │ 数据解析  │  │ 复杂计算  │  │ 图像处理  │           │    │
│  │  └──────────┘  └──────────┘  └──────────┘           │    │
│  └─────────────────────────────────────────────────────┘    │
└────────────────────────────────────────────────────────────┘
```

#### 实现代码

**主线程代码：**

```javascript
// main.js
class WorkerPool {
  constructor(workerScript, poolSize = navigator.hardwareConcurrency || 4) {
    this.poolSize = poolSize;
    this.workers = [];
    this.available = [];
    this.taskQueue = [];

    for (let i = 0; i < poolSize; i++) {
      const worker = new Worker(workerScript);
      worker.id = i;
      worker.busy = false;
      worker.onmessage = (e) => this._onComplete(worker, e);
      worker.onerror = (e) => this._onError(worker, e);
      this.workers.push(worker);
      this.available.push(worker);
    }
  }

  execute(data) {
    return new Promise((resolve, reject) => {
      const task = { data, resolve, reject };
      if (this.available.length > 0) {
        this._dispatch(this.available.shift(), task);
      } else {
        this.taskQueue.push(task);
      }
    });
  }

  _dispatch(worker, task) {
    worker.busy = true;
    worker._task = task;
    worker.postMessage(task.data);
  }

  _onComplete(worker, event) {
    const task = worker._task;
    worker.busy = false;
    worker._task = null;
    task.resolve(event.data);

    if (this.taskQueue.length > 0) {
      this._dispatch(worker, this.taskQueue.shift());
    } else {
      this.available.push(worker);
    }
  }

  _onError(worker, error) {
    worker._task.reject(error);
    worker.busy = false;
    worker._task = null;
    this.available.push(worker);
  }

  terminate() {
    this.workers.forEach((w) => w.terminate());
  }
}

// --- 使用示例 ---
const pool = new WorkerPool('./heavy-worker.js', 4);

pool.execute({ type: 'sort', data: largeArray })
  .then((result) => console.log('排序完成:', result));
```

**Worker 线程代码：**

```javascript
// heavy-worker.js
self.onmessage = function (e) {
  const { type, data } = e.data;

  switch (type) {
    case 'sort':
      const sorted = data.sort((a, b) => a - b);
      self.postMessage({ sorted, length: sorted.length });
      break;

    case 'filter':
      const filtered = data.filter((item) => item.active);
      self.postMessage({ filtered, length: filtered.length });
      break;

    case 'transform':
      const transformed = data.map((item) => ({
        ...item,
        processed: true,
        timestamp: Date.now(),
      }));
      self.postMessage({ transformed, length: transformed.length });
      break;

    default:
      self.postMessage({ error: 'Unknown task type' });
  }
};
```

#### 高并发场景适用性

- **大数据量排序/过滤**：10 万+ 条数据排序不阻塞 UI
- **Excel/CSV 解析**：大文件前端解析在 Worker 中执行
- **图片压缩/处理**：Canvas 操作在 Worker 中完成（OffscreenCanvas）
- **加密/解密**：敏感数据加解密不阻塞主线程

### 1.4 虚拟列表与懒加载

#### 原理

当需要渲染大量列表项（如万级数据）时，只渲染可视区域内的 DOM 节点，滚动时动态替换，大幅减少 DOM 节点数量。配合懒加载策略，按需加载数据和图片资源。

```
┌─────────────────────────────────┐
│  ┌─────────────────────────┐    │  ← 可视区域 (viewport)
│  │  Item 6                 │    │     仅渲染此区域的 DOM
│  │  Item 7  ← 实际渲染区域  │    │
│  │  Item 8                 │    │
│  │  Item 9                 │    │
│  │  Item 10                │    │
│  └─────────────────────────┘    │
│         ... 虚拟占位 ...          │  ← 上下用 padding 模拟滚动高度
└─────────────────────────────────┘
```

#### 实现代码

```javascript
/**
 * 高性能虚拟列表
 */
class VirtualList {
  constructor(options) {
    this.container = options.container;
    this.itemHeight = options.itemHeight || 50;
    this.bufferCount = options.bufferCount || 5;
    this.renderItem = options.renderItem;
    this.totalCount = 0;
    this.data = [];

    this._init();
  }

  _init() {
    // 创建视口容器
    this.viewport = document.createElement('div');
    this.viewport.style.cssText = `
      overflow-y: auto;
      height: 100%;
      position: relative;
    `;

    // 创建虚拟滚动区域
    this.scrollContent = document.createElement('div');
    this.scrollContent.style.position = 'relative';

    // 创建可见项容器
    this.visibleContent = document.createElement('div');
    this.visibleContent.style.position = 'absolute';
    this.visibleContent.style.top = '0';
    this.visibleContent.style.left = '0';
    this.visibleContent.style.right = '0';

    this.scrollContent.appendChild(this.visibleContent);
    this.viewport.appendChild(this.scrollContent);
    this.container.appendChild(this.viewport);

    // 绑定滚动事件（节流）
    this.viewport.addEventListener('scroll', throttle(() => this._render(), 16));
  }

  setData(data) {
    this.data = data;
    this.totalCount = data.length;
    this.scrollContent.style.height = `${this.totalCount * this.itemHeight}px`;
    this._render();
  }

  _render() {
    const scrollTop = this.viewport.scrollTop;
    const viewportHeight = this.viewport.clientHeight;

    // 计算可见范围
    const startIndex = Math.max(0, Math.floor(scrollTop / this.itemHeight) - this.bufferCount);
    const endIndex = Math.min(
      this.totalCount,
      Math.ceil((scrollTop + viewportHeight) / this.itemHeight) + this.bufferCount
    );

    // 更新可见区域位置
    this.visibleContent.style.transform = `translateY(${startIndex * this.itemHeight}px)`;

    // 渲染可见项
    const fragment = document.createDocumentFragment();
    for (let i = startIndex; i < endIndex; i++) {
      const item = this.renderItem(this.data[i], i);
      item.style.height = `${this.itemHeight}px`;
      item.style.position = 'absolute';
      item.style.top = `${(i - startIndex) * this.itemHeight}px`;
      fragment.appendChild(item);
    }

    this.visibleContent.innerHTML = '';
    this.visibleContent.appendChild(fragment);
  }
}

// --- 使用示例 ---
const virtualList = new VirtualList({
  container: document.getElementById('list-container'),
  itemHeight: 60,
  bufferCount: 8,
  renderItem: (item, index) => {
    const div = document.createElement('div');
    div.className = 'list-item';
    div.innerHTML = `<span>#${index}</span><span>${item.name}</span>`;
    return div;
  },
});

// 设置 10 万条数据
const bigData = Array.from({ length: 100000 }, (_, i) => ({
  id: i,
  name: `Item ${i}`,
}));
virtualList.setData(bigData);
```

#### 图片懒加载

```javascript
/**
 * 基于 IntersectionObserver 的图片懒加载
 */
class LazyLoader {
  constructor(options = {}) {
    this.rootMargin = options.rootMargin || '100px';
    this.threshold = options.threshold || 0.01;
    this.observer = null;
    this._init();
  }

  _init() {
    this.observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            const img = entry.target;
            const src = img.dataset.src;
            if (src) {
              img.src = src;
              img.removeAttribute('data-src');
            }
            this.observer.unobserve(img);
          }
        });
      },
      { rootMargin: this.rootMargin, threshold: this.threshold }
    );
  }

  observe(selector = '[data-src]') {
    document.querySelectorAll(selector).forEach((el) => {
      this.observer.observe(el);
    });
  }

  destroy() {
    this.observer.disconnect();
  }
}
```

### 1.5 服务端渲染 (SSR) 与静态生成 (SSG)

#### 原理

传统 SPA 需要下载 JS 后在客户端渲染，高并发下大量用户同时请求会导致首屏白屏时间长。SSR 在服务端完成 HTML 渲染后直接返回，SSG 在构建时预生成静态 HTML。

```
┌──────────────────────────────────────────────────────────────┐
│                        SSR 流程                              │
│                                                              │
│  客户端请求 ──→ Nginx/负载均衡 ──→ Node.js 服务端              │
│                                       │                      │
│                                       ▼                      │
│                               执行组件渲染 (React/Vue)        │
│                                       │                      │
│                                       ▼                      │
│                               生成完整 HTML 字符串             │
│                                       │                      │
│                                       ▼                      │
│              ←────── 返回 HTML + 脱水数据 ──────              │
│                                                              │
│  浏览器直接渲染 HTML ──→ 加载 JS 水合 ──→ 页面可交互           │
└──────────────────────────────────────────────────────────────┘
```

#### Vue3 SSR 实现示例

```javascript
// server.js - Express + Vue3 SSR
import { createSSRApp } from 'vue';
import { renderToString } from 'vue/server-renderer';
import express from 'express';

const app = express();

// 静态资源服务
app.use('/assets', express.static('dist/client'));

app.get('*', async (req, res) => {
  try {
    // 创建 Vue SSR 应用
    const vueApp = createSSRApp({
      template: `<div id="app">
        <h1>服务端渲染页面</h1>
        <p>当前路径: {{ path }}</p>
      </div>`,
      data() {
        return { path: req.url };
      },
    });

    // 渲染为 HTML 字符串
    const appContent = await renderToString(vueApp);

    // 注入到 HTML 模板
    const html = `
      <!DOCTYPE html>
      <html lang="zh-CN">
      <head>
        <meta charset="UTF-8">
        <meta name="viewport" content="width=device-width, initial-scale=1.0">
        <title>SSR Demo</title>
        <link rel="stylesheet" href="/assets/style.css">
      </head>
      <body>
        <div id="app">${appContent}</div>
        <script src="/assets/client.js"></script>
      </body>
      </html>
    `;

    res.setHeader('Content-Type', 'text/html');
    res.send(html);
  } catch (error) {
    console.error('SSR 渲染错误:', error);
    res.status(500).send('Server Error');
  }
});

app.listen(3000, () => {
  console.log('SSR 服务运行在 http://localhost:3000');
});
```

### 1.7 Service Worker 缓存与离线策略

#### 原理

Service Worker 是运行在浏览器后台的独立线程，可拦截网络请求，实现缓存策略、离线访问、推送通知等功能。高并发下可拦截重复请求，直接从缓存返回，减少服务端压力。

```
┌──────────────────────────────────────────────────────────────┐
│                  Service Worker 生命周期                       │
│                                                              │
│  注册 ──→ 安装 (install) ──→ 激活 (activate) ──→ 空闲/监听    │
│              │                    │              │            │
│              ▼                    ▼              ▼            │
│        预缓存关键资源        清理旧缓存      拦截 fetch 事件    │
│                                          ┌──────┴──────┐     │
│                                          ▼              ▼     │
│                                     缓存命中        网络请求    │
│                                     返回缓存        更新缓存    │
└──────────────────────────────────────────────────────────────┘
```

#### 实现代码

```javascript
// sw.js - Service Worker
const CACHE_NAME = 'app-v2';
const STATIC_ASSETS = [
  '/',
  '/index.html',
  '/css/app.css',
  '/js/app.js',
  '/js/vendor.js',
];

// 安装阶段：预缓存关键资源
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => {
      console.log('[SW] 预缓存关键资源');
      return cache.addAll(STATIC_ASSETS);
    })
  );
  // 跳过等待，立即激活
  self.skipWaiting();
});

// 激活阶段：清理旧缓存
self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((cacheNames) => {
      return Promise.all(
        cacheNames
          .filter((name) => name !== CACHE_NAME)
          .map((name) => caches.delete(name))
      );
    })
  );
  self.clients.claim();
});

// 请求拦截：缓存策略
self.addEventListener('fetch', (event) => {
  const { request } = event;
  const url = new URL(request.url);

  // API 请求：网络优先，失败时回退缓存
  if (url.pathname.startsWith('/api/')) {
    event.respondWith(networkFirst(request));
  }
  // 静态资源：缓存优先，后台更新
  else if (request.destination === 'script' || request.destination === 'style') {
    event.respondWith(cacheFirstWithRefresh(request));
  }
  // 图片：缓存优先
  else if (request.destination === 'image') {
    event.respondWith(cacheFirst(request));
  }
  // 其他：网络优先
  else {
    event.respondWith(networkFirst(request));
  }
});

// --- 缓存策略 ---

/** 缓存优先 */
async function cacheFirst(request) {
  const cached = await caches.match(request);
  if (cached) return cached;
  const response = await fetch(request);
  const cache = await caches.open(CACHE_NAME);
  cache.put(request, response.clone());
  return response;
}

/** 网络优先 */
async function networkFirst(request) {
  try {
    const response = await fetch(request);
    const cache = await caches.open(CACHE_NAME);
    cache.put(request, response.clone());
    return response;
  } catch {
    const cached = await caches.match(request);
    if (cached) return cached;
    throw new Error('Network unavailable');
  }
}

/** 缓存优先 + 后台更新 (Stale-While-Revalidate) */
async function cacheFirstWithRefresh(request) {
  const cached = await caches.match(request);
  const fetchPromise = fetch(request).then((response) => {
    caches.open(CACHE_NAME).then((cache) => {
      cache.put(request, response.clone());
    });
    return response;
  });
  return cached || fetchPromise;
}
```

### 2.2 首屏关键资源内联与预加载

#### 原理

将首屏渲染必需的关键 CSS 和 JS 内联到 HTML 中，避免额外的网络请求阻塞渲染。同时使用 `<link rel="preload">` 提前加载关键资源。

```
┌──────────────────────────────────────────────────────────────┐
│                    资源加载优先级策略                          │
│                                                              │
│  ┌──────────────────────────────────────────────────────┐    │
│  │ 阶段一：关键渲染路径 (Critical Rendering Path)         │    │
│  │  ├── 内联关键 CSS (<style>) — 无需请求，立即渲染        │    │
│  │  ├── 内联关键 JS 初始化代码 — 无需等待外部脚本           │    │
│  │  └── 首屏关键图片内联 (Base64) — 避免额外请求           │    │
│  └──────────────────────────────────────────────────────┘    │
│                           ▼                                  │
│  ┌──────────────────────────────────────────────────────┐    │
│  │ 阶段二：预加载 (Preload)                               │    │
│  │  ├── <link rel="preload"> 字体文件                    │    │
│  │  ├── <link rel="preload"> 首屏关键 JS                 │    │
│  │  └── <link rel="prefetch"> 下一页资源                  │    │
│  └──────────────────────────────────────────────────────┘    │
│                           ▼                                  │
│  ┌──────────────────────────────────────────────────────┐    │
│  │ 阶段三：异步加载 (Async/Defer)                         │    │
│  │  ├── <script defer> 非关键 JS                         │    │
│  │  ├── <link rel="stylesheet"> 非关键 CSS               │    │
│  │  └── 动态 import() 路由组件                           │    │
│  └──────────────────────────────────────────────────────┘    │
└──────────────────────────────────────────────────────────────┘
```

### 2.3 代码分割与路由懒加载

#### 原理

将应用代码按路由拆分为多个 chunk，用户访问某个路由时才加载对应代码，避免首屏加载全部 JS 导致白屏时间过长。Webpack/Vite 等构建工具原生支持动态 import 实现代码分割。

```
┌──────────────────────────────────────────────────────────────┐
│                    代码分割策略                               │
│                                                              │
│  构建产物:                                                    │
│  ┌────────────┐  ┌────────────┐  ┌────────────┐             │
│  │  main.js   │  │ vendor.js  │  │ common.js  │  ← 首屏加载  │
│  │  (入口)    │  │ (框架库)   │  │ (公共代码)  │             │
│  └────────────┘  └────────────┘  └────────────┘             │
│                                                              │
│  ┌────────────┐  ┌────────────┐  ┌────────────┐             │
│  │ Home.js    │  │ About.js   │  │ Admin.js   │  ← 按需加载  │
│  │ (首页)     │  │ (关于页)   │  │ (管理后台)  │             │
│  └────────────┘  └────────────┘  └────────────┘             │
│                                                              │
```

### 2.4 预渲染 (Pre-rendering)

#### 原理

在构建时使用无头浏览器将指定路由的页面渲染为静态 HTML 文件，部署后直接返回给用户。不需要 Node.js 服务端运行，适合内容不频繁变化的页面。

```
┌──────────────────────────────────────────────────────────────┐
│                     预渲染流程                               │
│                                                              │
│  构建阶段:                                                    │
│  ┌──────────┐    ┌──────────────┐    ┌──────────────┐        │
│  │ 源代码    │ →  │ 无头浏览器     │ →  │ 静态 HTML    │        │
│  │ (Vue/     │    │ (Puppeteer)   │    │ /index.html  │        │
│  │  React)   │    │ 渲染每个路由   │    │ /about.html  │        │
│  └──────────┘    └──────────────┘    │ /product/    │        │
│                                       │   index.html │        │
│                                       └──────────────┘        │
│                                                              │
│  部署后:                                                      │
│  ┌──────────┐    ┌──────────────┐    ┌──────────────┐        │
│  │ 用户请求  │ →  │ Nginx/CDN    │ →  │ 返回静态 HTML │        │
│  │ /about   │    │ 直接返回文件  │    │ (瞬时响应)    │        │
│  └──────────┘    └──────────────┘    └──────────────┘        │
│                                              │               │
│                                              ▼               │
│                                      客户端 JS 水合           │
│                                      (页面变为可交互)          │
└──────────────────────────────────────────────────────────────┘
```

#### 实现代码

**Vite 预渲染配置：**

```javascript
// vite.config.js
import { defineConfig } from 'vite';
import vue from '@vitejs/plugin-vue';
import { vitePrerenderPlugin } from 'vite-plugin-prerender';
import path from 'path';

export default defineConfig({
  plugins: [
    vue(),
    vitePrerenderPlugin({
      // 指定需要预渲染的路由
      routes: ['/', '/about', '/products', '/contact'],

      // 静态文件输出目录
      staticDir: path.resolve(__dirname, 'dist'),

      // 渲染器配置
      renderer: new vitePrerenderPlugin.PuppeteerRenderer({
        // 等待网络请求完成
        renderAfterDocumentEvent: 'prerender-ready',
        // 超时设置
        timeout: 30000,
        // 注入脚本
        inject: {
          // 在所有页面注入
          prerender: true,
        },
      }),

      // 后处理：对生成的 HTML 进行优化
      postProcess(renderedRoute) {
        // 移除预渲染标记
        renderedRoute.html = renderedRoute.html.replace(
          /<script[^>]*prerender[^>]*>.*?<\/script>/g,
          ''
        );
        return renderedRoute;
      },
    }),
  ],
});
```

### 2.5 错误兜底与优雅降级

#### 原理

JS 执行错误是导致白屏的重要原因之一。React/Vue 等框架渲染过程中如果抛出未捕获的异常，会导致整个组件树卸载，出现白屏。通过错误边界、全局错误捕获、降级渲染等机制，确保即使出错也能展示有意义的内容。

#### 实现代码

**Vue3 全局错误处理：**

```javascript
// main.js
import { createApp } from 'vue';
import App from './App.vue';

const app = createApp(App);

// 全局错误处理器
app.config.errorHandler = (err, instance, info) => {
  console.error('[Vue Error]', err, info);

  // 上报错误
  reportError({
    message: err.message,
    stack: err.stack,
    component: instance?.$options?.name || 'unknown',
    info,
    url: window.location.href,
    timestamp: Date.now(),
  });

  // 不阻止默认行为，让错误边界组件处理
};

// 全局警告处理器
app.config.warnHandler = (msg, instance, trace) => {
  console.warn('[Vue Warning]', msg, trace);
};

app.mount('#app');

// 未捕获的 Promise 错误
window.addEventListener('unhandledrejection', (event) => {
  console.error('[Unhandled Promise Rejection]', event.reason);
  reportError({
    message: event.reason?.message || 'Promise rejection',
    stack: event.reason?.stack,
    type: 'unhandledrejection',
  });
  event.preventDefault();
});
```

```vue
  return false;
});

  hasError.value = false;
  errorInfo.value = null;
}
</script>
```

#### 资源加载失败降级

```javascript
/**
 * CDN 资源加载失败自动降级
 */
class ResourceFallback {
  static init() {
    this._handleScriptError();
    this._handleLinkError();
  }

  // JS 加载失败降级
  static _handleScriptError() {
    window.addEventListener('error', (e) => {
      const target = e.target;
      if (target.tagName === 'SCRIPT' && target.src) {
        const fallback = target.dataset.fallback;
        if (fallback) {
          console.warn(`[ResourceFallback] JS 加载失败: ${target.src}, 降级: ${fallback}`);
          const script = document.createElement('script');
          script.src = fallback;
          document.head.appendChild(script);
        }
      }
    }, true);
  }

  // CSS 加载失败降级
  static _handleLinkError() {
    window.addEventListener('error', (e) => {
      const target = e.target;
      if (target.tagName === 'LINK' && target.href) {
        const fallback = target.dataset.fallback;
        if (fallback) {
          console.warn(`[ResourceFallback] CSS 加载失败: ${target.href}, 降级: ${fallback}`);
          const link = document.createElement('link');
          link.rel = 'stylesheet';
          link.href = fallback;
          document.head.appendChild(link);
        }
      }
    }, true);
  }
}

ResourceFallback.init();
```

- **复杂交互页面**：局部错误不影响其他区域
