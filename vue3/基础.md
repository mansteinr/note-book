### 组合式函数--请求封装

```
import { reactive, toRefs } from 'vue'
import axios from 'axios'
interface IResult<T> {
  result: null | T;
  loading: boolean;
  error: any;
}
const useURLLoader = <T = any>(url: string) => {
  const data:IResult<T> = reactive({
    result: null,
    loading: true,
    error: null
  })
  axios.get(url).then(rawData => {
    data.result = rawData.data
  }).catch(e => {
    data.error = e
  }).finally(() => {
    data.loading = false
  })
  return toRefs(data)
}

export default useURLLoader
```


在模板中使用

```
import useURLLoader from './hooks/useURLLoader'

// 因为返回的结构 为{
//  id: 12312,
//  title: 'this is a title'
// } 

interface TodoResult {
  title: string;
}

const { loading, result } = useURLLoader<TodoResult>('https://jsonplaceholder.typicode.com/todos/1')


<h1 v-if="loading">Loading...</h1>
<h1 v-if="!loading && result">{{result.title}}</h1>
```
